cmake_minimum_required(VERSION 3.16)
project(supermimic_imu_raw_reader)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
# Match the installed SDK's existing imu-compare build include-path correction.
get_property(imported DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(target IN LISTS imported)
  get_target_property(incs ${target} INTERFACE_INCLUDE_DIRECTORIES)
  if(incs)
    string(REPLACE "/usr/local/lib/python3.10/dist-packages/numpy/core/include" "/usr/lib/python3/dist-packages/numpy/core/include" incs "${incs}")
    set_target_properties(${target} PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${incs}")
  endif()
endforeach()
add_executable(imu_raw_reader reader.cpp)
target_compile_features(imu_raw_reader PRIVATE cxx_std_17)
ament_target_dependencies(imu_raw_reader rclcpp sensor_msgs)
